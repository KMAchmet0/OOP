class CustomArrayList  {
   int n = 5;
   class ReceivedList {
      
      String received;
      ReceivedList(String received) {
         this.received =received ;
      }
   }
   public static void main(String args[]) {
      String received[] = {"Food", "Wood", "Clothes", "Gloves"};
      CustomArrayList caList = new CustomArrayList();
      caList.add(received);
   }
   public void add( String received[]) {
      ArrayList<ReceivedList> aList = new ArrayList<>();
      for (int i = 0; i < n; i++) {
         aList.add(new ReceivedList(received[i]));
      }
      print(aList);
   }
   public void print(ArrayList<ReceivedList> aList) {
      for (int i = 0; i < n; i++) {
         ReceivedList r = aList.get(i);
         System.out.println("\nReceived products are: " + r.received);
         
      }
   }
